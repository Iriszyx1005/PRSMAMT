make_option("--train_out", action="store", default="/home/sshen/Disk_m2/PRS/LungCa_V3/PRSCSx", type='character',
help="Path to output files [required]"),
make_option("--train_pheno", action="store", default="/home/sshen/Disk_m2/PRS_yxzhang/train/asian/pheno_1974", type='character',help="pheno [required]"),

  make_option("--cohort", action="store", default="oncoarray_eas", type='character',help="trait [required]"),
make_option("--trait", action="store", default="LungCa", type='character',help="trait [required]")
)
opt = parse_args(OptionParser(option_list=option_list))
setwd(opt$train_out)

score=NULL
for(i in 1:22){
  if (file.exists(paste0(opt$train_out,"/",opt$train,"/",opt$train,"_",opt$cohort,i,".sscore"))) {
    x=fread(paste0(opt$train_out,"/",opt$train,"/",opt$train,"_",opt$cohort,i,".sscore"))
    if(colnames(x)[1]=="#FID") x = x[,-1]
    colnames(x)[1] = "IID"
    y=x[,"IID"]
    #x$SCORE1_SUM=x$SCORE1_AVG*x$ALLELE_CT
    x=x[,5]
    setnames(x,c("SCORE1_SUM"),c(paste0("chr",i)))
    score=cbind(score,x)
  }
}

score$PRS<- rowSums(score)

score=cbind(y,score)
cols=c("IID","PRS")
score=score[,..cols]
write.table(score,paste0(opt$train_out,"/",opt$train,"/",opt$train,"_",opt$cohort,"_prs.txt"),row.names = F,quote=F,sep="\t")

covar_onco = fread(paste0(opt$train_pheno,".txt"))
prs = merge(score,covar_onco,by="IID");prs[,2] = apply(prs[,2],2,scale)


model = glm(Lungcancer~PRS,family = binomial,data = prs)
 auc(roc(prs$Lungcancer,prs$PRS))
exp(coef(model)[2])
coef(summary(model))[2,4]
