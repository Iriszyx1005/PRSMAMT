```sh
cd /home/sshen/Disk_m2/PRS_yxzhang/LDScore/popcorn
unzip g1000_eur.zip
unzip g1000_eas.zip
unzip g1000_afr.zip


# AFR population
#cat 1000GP_Phase3.sample | grep YRI | awk '{print $1 " " $1}' > YRI.samples
~/software/plink2 --bfile g1000_afr \
  --chr 6 --from-mb 25 --to-mb 35 \
  --make-just-bim \
  --out mhc_afr --threads 1

awk '{print $2}' mhc_afr.bim > mhc_afr.justrs

~/software/plink --bfile g1000_afr --exclude mhc_afr.justrs --maf 0.01 --make-bed --out g1000_afr_noMHC_maf0p01 

# EUR population
~/software/plink2 --bfile g1000_eur \
  --chr 6 --from-mb 25 --to-mb 35 \
  --make-just-bim \
  --out mhc_eur --threads 1

cat mhc_eur.bim | awk '{print $2}' > mhc_eur.justrs

~/software/plink --bfile g1000_eur  --exclude mhc_eur.justrs --maf 0.01 --make-bed --out g1000_eur_noMHC_maf0p01  


# EAS population
~/software/plink2 --bfile g1000_eas \
  --chr 6 --from-mb 25 --to-mb 35 \
  --make-just-bim \
  --out mhc_eas --threads 1

cat mhc_eas.bim | awk '{print $2}' > mhc_eas.justrs

~/software/plink --bfile g1000_eas  --exclude mhc_eas.justrs --maf 0.01 --make-bed --out g1000_eas_noMHC_maf0p01  

nohup popcorn compute --SNPs_to_store 50000 --bfile g1000_eur_noMHC_maf0p01 EUR.cscore &

nohup popcorn compute --SNPs_to_store 50000 --bfile g1000_eas_noMHC_maf0p01 EAS.cscore &

nohup popcorn compute --SNPs_to_store 50000 --bfile g1000_afr_noMHC_maf0p01 AFR.cscore &

nohup popcorn compute --SNPs_to_store 50000 --bfile1 g1000_eur_noMHC_maf0p01  --bfile2 g1000_afr_noMHC_maf0p01 EUR_AFR_all_gen_eff.cscore &

nohup popcorn compute --SNPs_to_store 50000 --bfile1 g1000_eas_noMHC_maf0p01  --bfile2 g1000_afr_noMHC_maf0p01 EAS_AFR_all_gen_eff.cscore &


# for the bivariate (cross-trait)
popcorn fit --sfile1 data/TRAIT_EUR.sumstats --sfile2 data/TRAIT_AAM.sumstats  --cfile refs/g1000_eur_asw.cscore results/TRAIT_EUR_AAM.out
```

```R
library(data.table)
library(dplyr)
library(ggplot2)
library(RColorBrewer)
setwd("/home/sshen/Disk_m2/PRS_yxzhang/gene_cor/result")
a=dir()
s <- a[grep(".txt$", a)]
re=NULL
for (i in s){
  x=fread(i)
  label=gsub(".txt", "",i)
  Cor=x[x$V1=="pgi",]$'Val (obs)'
  SE =x[x$V1=="pgi",]$SE  
  Z=x[x$V1=="pgi",]$Z  
  P=x[x$V1=="pgi",]$'P (Z)'
 re=rbind(re,data.frame(label,Cor,SE,Z,P))
}
write.table(re,paste0("gene_cor.txt"),row.names = F,quote=F,sep="\t")
```
